cmake_minimum_required(VERSION 3.16)
project(C7_BrickGame C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)

file(GLOB BACKEND_SOURCES "src/brick_game/tetris/*.c")
add_library(s21_tetris STATIC ${BACKEND_SOURCES})
target_include_directories(s21_tetris PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src/brick_game/tetris/inc
)

enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(CHECK REQUIRED check)

file(GLOB TEST_SOURCES "src/brick_game/tests/*.c")

add_executable(run_game_tests src/brick_game/tests/tests_main.c ${TEST_SOURCES})

target_include_directories(run_game_tests PRIVATE ${CHECK_INCLUDE_DIRS})
target_link_libraries(run_game_tests PRIVATE s21_tetris ${CHECK_LIBRARIES})

find_package(Curses REQUIRED)

add_executable(s21_BrickGame
    src/s21_main.c
    src/gui/cli/s21_frontend.c
)
target_include_directories(s21_BrickGame PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/gui/cli/inc
)
target_link_libraries(s21_BrickGame PRIVATE s21_tetris ${CURSES_LIBRARIES})


set(MAKE_COMMAND make -C src)

add_custom_target(make_all
        COMMAND ${MAKE_COMMAND} all
        COMMENT "Building and installing the game using Makefile (make all)"
        VERBATIM
)

add_custom_target(make_install
        COMMAND ${MAKE_COMMAND} install
        COMMENT "Installing the game using Makefile (make install)"
        VERBATIM
)

add_custom_target(make_test
        COMMAND ${MAKE_COMMAND} test
        COMMENT "Running tests using Makefile (make test)"
        VERBATIM
)

add_custom_target(make_dist
        COMMAND ${MAKE_COMMAND} dist
        COMMENT "Creating distribution package using Makefile (make dist)"
        VERBATIM
)

add_custom_target(make_gcov_report
        COMMAND ${MAKE_COMMAND} gcov_report
        COMMENT "Generating GCOV report using Makefile (make gcov_report)"
        VERBATIM
)

add_custom_target(make_format_style
        COMMAND ${MAKE_COMMAND} format_style
        COMMENT "Formatting code using Makefile (make format_style)"
        VERBATIM
)

add_custom_target(make_style_check
        COMMAND ${MAKE_COMMAND} style_check
        COMMENT "Checking code style using Makefile (make style_check)"
        VERBATIM
)

add_custom_target(make_clean
        COMMAND ${MAKE_COMMAND} clean
        COMMENT "Cleaning build artifacts using Makefile (make clean)"
        VERBATIM
)
